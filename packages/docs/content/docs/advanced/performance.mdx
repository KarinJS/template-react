模板项目的性能优化涉及构建体积、渲染速度、CSS 大小和开发体验。本页讲解实用优化技巧和实际案例。

## 构建体积优化

### Tree-shaking

确保只打包使用的代码：

```ts title="karin.template.ts" twoslash
import { defineConfig } from '@karinjs/template-react'
// ---cut---
export default defineConfig({
  vite: {
    build: {
      rollupOptions: {
        treeshake: {
          moduleSideEffects: false // 移除无副作用的未使用代码
        }
      }
    }
  }
})
```

**具名导入而非全量导入**：

```tsx twoslash
// ✅ 好：tree-shakable
import { Button, Card } from '@heroui/react'

export const good = <Button>好的写法</Button>
```

```tsx twoslash
// ❌ 差：命名空间导入，打包器无法 tree-shake，整库进产物
import * as HeroUI from '@heroui/react'

export const bad = <HeroUI.Button>差的写法</HeroUI.Button>
```

### 动态导入

大组件或第三方库按需加载：

```tsx title="ktr/template/chart/stats/index.tsx" twoslash
import { defineTemplate, type TemplateProps } from '@karinjs/template-react'
import { lazy, Suspense } from 'react'

// 懒加载图表库
const Chart = lazy(() => import('react-chartjs-2').then((m) => ({ default: m.Line })))

interface StatsData {
  labels: string[]
  values: number[]
}

const Stats = ({ data }: TemplateProps<StatsData>) => (
  <Suspense fallback={<div>加载中...</div>}>
    <Chart data={{ labels: data.labels, datasets: [{ data: data.values }] }} />
  </Suspense>
)

export default defineTemplate({
  component: Stats
})
```

## 渲染速度优化

### 组件懒加载

仅在使用时加载组件：

```tsx title="ktr/template/dashboard/index.tsx" twoslash
// @filename: ../../components/HeavyChart.tsx
export default function HeavyChart() {
  return <div>chart</div>
}
// @filename: ../../components/HeavyTable.tsx
export default function HeavyTable() {
  return <div>table</div>
}
// @filename: index.tsx
// ---cut---
import { defineTemplate, type TemplateProps } from '@karinjs/template-react'
import { lazy, Suspense } from 'react'

const HeavyChart = lazy(() => import('../../components/HeavyChart'))
const HeavyTable = lazy(() => import('../../components/HeavyTable'))

interface DashboardData {
  showChart: boolean
  showTable: boolean
}

const Dashboard = ({ data }: TemplateProps<DashboardData>) => (
  <div>
    {data.showChart && (
      <Suspense fallback={<div>加载图表...</div>}>
        <HeavyChart />
      </Suspense>
    )}
    {data.showTable && (
      <Suspense fallback={<div>加载表格...</div>}>
        <HeavyTable />
      </Suspense>
    )}
  </div>
)

export default defineTemplate({
  component: Dashboard
})
```

### React.memo 避免重渲染

```tsx title="ktr/template/components/UserCard.tsx" twoslash
import React from 'react'

interface UserCardProps {
  name: string
  avatar: string
}

export const UserCard = React.memo<UserCardProps>(({ name, avatar }) => {
  console.log('UserCard 渲染')

  return (
    <div className="flex items-center gap-3">
      <img src={avatar} alt={name} className="size-12 rounded-full" />
      <span className="font-medium">{name}</span>
    </div>
  )
})
```

### useMemo 缓存计算

```tsx twoslash
import { useMemo } from 'react'

interface Item {
  score: number
}
interface Props {
  items: Item[]
}

function Component({ items }: Props) {
  const total = useMemo(() => {
    console.log('计算总分')
    return items.reduce((sum, item) => sum + item.score, 0)
  }, [items])

  return <div>总分: {total}</div>
}
```

### 虚拟列表

长列表使用 react-window：

```tsx title="ktr/template/user/list/index.tsx" twoslash
import { defineTemplate, type TemplateProps } from '@karinjs/template-react'

  component: UserList
})
```

## 实际优化案例

### 案例 1：减少首屏体积 70%

**问题**：首次加载需要下载 800KB JS。

**优化**：

```ts twoslash
import { defineConfig } from '@karinjs/template-react'
import { visualizer } from 'rollup-plugin-visualizer'

export default defineConfig({
  vite: {
    plugins: [
      visualizer({ open: true }) // 1. 分析体积
    ],
    build: {
      rollupOptions: {
        output: {
          manualChunks(id) {
            // 2. 拆分第三方库（vite 8 / rolldown 的 manualChunks 是函数形式）
            if (id.includes('node_modules/react')) return 'vendor'
            if (id.includes('@heroui')) return 'ui'
          }
        }
      }
    }
  }
})
```

```tsx twoslash
// @filename: ./HeavyChart.tsx
export default function HeavyChart() {
  return <div>chart</div>
}
// @filename: index.tsx
// ---cut---
import { lazy } from 'react'

// 3. 懒加载重组件
const HeavyChart = lazy(() => import('./HeavyChart'))
```

**结果**：首屏减少到 240KB。

### 案例 2：优化大列表渲染

**问题**：渲染 1000 个用户卡片卡顿。

**优化**：

```tsx title="ktr/template/user/list/index.tsx" twoslash
import { defineTemplate, type TemplateProps } from '@karinjs/template-react'
import React from 'react'
import { List, type RowComponentProps } from 'react-window'

interface User {
  id: string
  name: string
  avatar: string
}

interface ListData {
  users: User[]
}

// 1. memo 避免重渲染
const UserRow = React.memo<{ user: User }>(({ user }) => (
  <div className="flex items-center gap-3 px-4 py-2">
    <img src={user.avatar} alt="" className="size-10 rounded-full" />
    <span>{user.name}</span>
  </div>
))

// 2. 虚拟列表
const Row = ({ index, style, users }: RowComponentProps<{ users: User[] }>) => (
  <div style={style}>
    <UserRow user={users[index]} />
  </div>
)

const UserList = ({ data }: TemplateProps<ListData>) => (
  <List rowComponent={Row} rowCount={data.users.length} rowHeight={56} rowProps={{ users: data.users }} style={{ height: 600 }} />
)

export default defineTemplate({
  component: UserList
})
```

**结果**：渲染时间从 2.5s 降到 80ms。

### 案例 5：SSR 渲染性能

**问题**：单个模板 SSR 渲染耗时 800ms。

**优化**：

```tsx twoslash
import { defineTemplate, type TemplateProps } from '@karinjs/template-react'
import { useMemo } from 'react'

interface Data {
  users: Array<{ id: string; score: number }>
}

const Template = ({ data }: TemplateProps<Data>) => {
  // 1. 缓存计算结果
  const sortedUsers = useMemo(() => [...data.users].sort((a, b) => b.score - a.score), [data.users])

  return (
    <div>
      {sortedUsers.map((user) => (
        // 2. 使用 key 优化 reconciliation
        <div key={user.id}>{user.score}</div>
      ))}
    </div>
  )
}

export default defineTemplate({
  component: Template
})
```

```ts twoslash
import type { RenderPlugin } from '@karinjs/template-react'

// 3. 性能监控插件
const perfPlugin: RenderPlugin = {
  name: 'perf',
  beforeRender: (ctx) => {
    ;(ctx as any).__startTime = Date.now()
  },
  afterRender: (ctx) => {
    const duration = Date.now() - (ctx as any).__startTime
    if (duration > 500) {
      console.warn(`[慢渲染] ${ctx.path}: ${duration}ms`)
    }
  }
}
```

**结果**：渲染时间降到 120ms。

### 案例 6：图片优化

**问题**：模板包含大量高清图片，加载慢。

**优化**：

```tsx twoslash
import React from 'react'

// 1. 响应式图片
const Avatar: React.FC<{ src: string; alt: string }> = ({ src, alt }) => (
  <picture>
    <source srcSet={`${src}?w=64&format=avif 1x, ${src}?w=128&format=avif 2x`} type="image/avif" />
    <source srcSet={`${src}?w=64&format=webp 1x, ${src}?w=128&format=webp 2x`} type="image/webp" />
    <img src={`${src}?w=64`} alt={alt} loading="lazy" decoding="async" className="size-16 rounded-full" />
  </picture>
)
```

```ts twoslash
import type { RenderPlugin } from '@karinjs/template-react'
import fs from 'node:fs'
import path from 'node:path'

// 2. 小图片内联（框架已内置该能力：html.assetsInlineLimit 默认 4096 字节，
//    需要更定制的规则时才用这种 afterRender 插件，比如按路径前缀区分阈值）
const inlineSmallImages: RenderPlugin = {
  name: 'inline-small-images',
  afterRender: (ctx) => {
    return ctx.html.replace(/<img[^>]+src="\/assets\/([^"]+\.(?:png|jpg|svg))"[^>]*>/g, (match, filename) => {
      const filepath = path.join(ctx.outputDir, '../assets', filename)
      if (!fs.existsSync(filepath)) return match

      const stat = fs.statSync(filepath)
      if (stat.size > 5120) return match // 只内联 < 5KB

      const ext = path.extname(filename).slice(1)
      const base64 = fs.readFileSync(filepath, 'base64')
      return match.replace(`src="/assets/${filename}"`, `src="data:image/${ext};base64,${base64}"`)
    })
  }
}
```

**结果**：图片加载时间减少 80%。

### 案例 7：减少重渲染

**问题**：父组件更新导致子组件不必要的重渲染。

**优化**：

```tsx twoslash
import React, { useState } from 'react'

interface User {
  id: string
  name: string
}

// 1. memo + 原始值 props
const UserCard = React.memo<{ user: User }>(({ user }) => {
  console.log('UserCard 渲染')
  return <div>{user.name}</div>
})

// 2. 回调使用 useCallback
const UserList: React.FC<{ users: User[] }> = ({ users }) => {
  const [selected, setSelected] = useState<string | null>(null)

  const handleSelect = React.useCallback((id: string) => {
    setSelected(id)
  }, [])

  return (
    <div>
      {users.map((user) => (
        <UserCard key={user.id} user={user} />
      ))}
    </div>
  )
}
```

**结果**：更新时只重渲染必要的组件。
