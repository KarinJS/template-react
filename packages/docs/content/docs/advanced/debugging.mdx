## SSR 渲染调试

### 渲染日志

在插件中记录渲染过程：

```ts twoslash
import type { RenderPlugin } from '@karinjs/template-react'

const debugPlugin: RenderPlugin = {
  name: 'debug',
  beforeRender: (ctx) => {
    console.log('[SSR] 开始渲染:', ctx.path)
    console.log('[SSR] 数据:', JSON.stringify(ctx.data, null, 2))
    console.log('[SSR] 主题:', ctx.ctx.theme)
  },
  afterRender: (ctx) => {
    console.log('[SSR] HTML 大小:', ctx.html.length)
    console.log('[SSR] 包含的类名:', extractClassNames(ctx.html))
  }
}

function extractClassNames(html: string): string[] {
  const matches = html.matchAll(/class="([^"]*)"/g)
  const classes = new Set<string>()
  for (const match of matches) {
    match[1].split(/\s+/).forEach((cls) => cls && classes.add(cls))
  }
  return Array.from(classes)
}
```

### 错误捕获

捕获 SSR 渲染错误：

```ts twoslash
import { createRenderer } from '@karinjs/template-react'
import type { RenderPlugin } from '@karinjs/template-react'

const errorPlugin: RenderPlugin = {
  name: 'error-catcher',
  beforeRender: async (ctx) => {
    try {
      // 数据校验
      if (!ctx.data) throw new Error('数据为空')
    } catch (error) {
      console.error(`[SSR 错误] ${ctx.path}:`, error)
      // 记录到文件
      await logError(ctx.path, error)
      throw error
    }
  }
}

async function logError(path: string, error: unknown) {
  const fs = await import('node:fs/promises')
  const timestamp = new Date().toISOString()
  const log = `[${timestamp}] ${path}: ${(error as Error).message}\n`
  await fs.appendFile('ssr-errors.log', log)
}
```

### React 错误边界

在 SSR 中捕获组件错误：

```tsx title="ktr/template/components/ErrorBoundary.tsx" twoslash
import React from 'react'

interface Props {
  children: React.ReactNode
}

interface State {
  hasError: boolean
  error?: Error
}

export class ErrorBoundary extends React.Component<Props, State> {
  constructor(props: Props) {
    super(props)
    this.state = { hasError: false }
  }

  static getDerivedStateFromError(error: Error): State {
    return { hasError: true, error }
  }

  componentDidCatch(error: Error, errorInfo: React.ErrorInfo) {
    console.error('[ErrorBoundary] 捕获错误:', error, errorInfo)
  }

  render() {
    if (this.state.hasError) {
      return (
        <div className="p-6 bg-error-soft text-error">
          <h2 className="text-lg font-bold mb-2">渲染错误</h2>
          <pre className="text-sm">{this.state.error?.message}</pre>
        </div>
      )
    }

    return this.props.children
  }
}
```

使用错误边界：

```tsx title="ktr/template/user/card/index.tsx" twoslash
import { defineTemplate, type TemplateProps } from '@karinjs/template-react'
import React from 'react'

interface Props {
  children: React.ReactNode
}

interface State {
  hasError: boolean
  error?: Error
}

class ErrorBoundary extends React.Component<Props, State> {
  constructor(props: Props) {
    super(props)
    this.state = { hasError: false }
  }

  static getDerivedStateFromError(error: Error): State {
    return { hasError: true, error }
  }

  render() {
    if (this.state.hasError) {
      return <div>Error</div>
    }
    return this.props.children
  }
}

interface Data {
  name: string
}

const UserCard = ({ data }: TemplateProps<Data>) => (
  <ErrorBoundary>
    <div>{data.name}</div>
  </ErrorBoundary>
)

export default defineTemplate({
  component: UserCard
})
```

## 常见调试场景

### 场景 1：模板未注册

**问题**：开发面板看不到新建的模板。

**调试步骤**：

```bash
# 1. 检查文件位置
ls -la ktr/template/user/card/
# 应包含 index.tsx

# 2. 手动刷新注册表
pnpm ktr sync

# 3. 查看生成的注册表
cat .ktr/template-registry.ts
# 检查是否包含 'user/card'

# 4. 检查是否被忽略
# 文件名不是 index.tsx？
# 目录名以 _ 开头？
# 在 components/ 下？
```

**解决方案**：

```tsx title="ktr/template/user/card/index.tsx" twoslash
import { defineTemplate } from '@karinjs/template-react'

// ✅ 正确：导出 defineTemplate
export default defineTemplate({
  component: () => <div>User Card</div>
})

// ❌ 错误：直接导出组件
// export default () => <div>User Card</div>
```

### 场景 2：数据类型不匹配

**问题**：运行时数据与类型定义不一致。

**调试步骤**：

```tsx title="ktr/template/user/card/index.tsx" twoslash
import { defineTemplate, type TemplateProps } from '@karinjs/template-react'

interface UserCardData {
  name: string
  age: number
}

const UserCard = ({ data }: TemplateProps<UserCardData>) => {
  // 1. 运行时检查
  console.log('数据类型:', typeof data)
  console.log('数据内容:', data)
  console.log('是否有 age:', 'age' in data)

  return <div>{data.name}</div>
}

export default defineTemplate({
  component: UserCard,
  // 2. 添加 validate
  validate: (data): data is UserCardData => {
    if (typeof data !== 'object' || data === null) {
      console.error('[validate] 数据不是对象:', data)
      return false
    }
    if (!('name' in data) || typeof (data as any).name !== 'string') {
      console.error('[validate] 缺少 name 字段或类型错误')
      return false
    }
    if (!('age' in data) || typeof (data as any).age !== 'number') {
      console.error('[validate] 缺少 age 字段或类型错误')
      return false
    }
    return true
  }
})
```

### 场景 4：图片加载失败

**问题**：开发环境图片正常，SSR 渲染后图片 404。

**调试步骤**：

```tsx title="ktr/template/user/card/index.tsx" twoslash
import { defineTemplate, type TemplateProps } from '@karinjs/template-react'

interface Data {
  avatar: string
}

const Component = ({ data }: TemplateProps<Data>) => {
  // 1. 打印图片路径
  console.log('图片路径:', data.avatar)

  // 2. 检查路径格式
  // ✅ 正确：/assets/avatar.png
  // ❌ 错误：./assets/avatar.png
  // ❌ 错误：assets/avatar.png

  return <img src={data.avatar} alt="头像" />
}

export default defineTemplate({ component: Component })
```

**解决方案**：

```tsx
// ✅ 正确：使用 / 开头的路径（等于 <dir.assets>/ 下的相对路径）
<img src="/assets/avatar.png" alt="" />

// ❌ 错误：使用相对路径
<img src="./avatar.png" alt="" />
```

SSR 产出 HTML 时框架会自动改写这些引用：不超过 `html.assetsInlineLimit`（默认 4096 字节）的内联为 base64，超过的转为 `file://` 绝对路径，开发和生产都不需要手写路径转换插件。如果图片仍然 404，按顺序检查：

1. 引用路径是否以 `/` 开头（`./assets/x.png`、`assets/x.png` 都不会被处理）；
2. 文件是否真实存在于 `<dir.assets>/` 下的对应位置（如 `ktr/public/assets/avatar.png`）；
3. 生产态确认产物目录下 `assets/` 存在（如 `lib/assets/`，由构建时的 `copyAssets` 复制）。

### 场景 8：插件不执行

## 调试配置

### 启用详细日志

```ts title="karin.template.ts" twoslash
import { defineConfig } from '@karinjs/template-react'
// ---cut---
export default defineConfig({
  vite: {
    logLevel: 'info', // 'error' | 'warn' | 'info' | 'silent'
    clearScreen: false // 不清屏，保留历史日志
  }
})
```

### Source Map

```ts title="karin.template.ts" twoslash
import { defineConfig } from '@karinjs/template-react'
// ---cut---
export default defineConfig({
  vite: {
    build: {
      sourcemap: true // 生成 source map 用于调试
    }
  }
})
```

### 错误堆栈

```ts title="src/utils/render.ts"
Error.stackTraceLimit = 50 // 增加堆栈深度
```
