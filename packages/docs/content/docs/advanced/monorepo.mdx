在 Monorepo 中管理多个模板包，复用组件、样式和类型，同时保持各包独立构建。

## Monorepo 场景

### 场景 1：多机器人共享模板

```
workspace/
├── packages/
│   ├── shared/               # 共享包
│   │   ├── templates/        # 共享模板
│   │   ├── components/       # 通用组件
│   │   ├── styles/           # 共享样式
│   │   └── types/            # 类型定义
│   ├── bot-a/                # 机器人 A
│   │   └── karin.template.ts
│   └── bot-b/                # 机器人 B
│       └── karin.template.ts
├── pnpm-workspace.yaml
└── package.json
```

### 场景 2：按功能模块拆分

```
workspace/
├── packages/
│   ├── core/                 # 核心模板
│   ├── games/                # 游戏相关
│   ├── social/               # 社交功能
│   └── admin/                # 管理后台
└── pnpm-workspace.yaml
```

### 场景 3：多租户 SaaS

```
workspace/
├── packages/
│   ├── base/                 # 基础模板
│   ├── tenant-a/             # 租户 A 定制
│   └── tenant-b/             # 租户 B 定制
└── pnpm-workspace.yaml
```

## pnpm Workspace 配置

### 初始化 Workspace

```yaml title="pnpm-workspace.yaml"
packages:
  - 'packages/*'
  - 'apps/*'
```

### 根 package.json

```json title="package.json"
{
  "name": "karin-templates-workspace",
  "private": true,
  "scripts": {
    "dev": "pnpm --filter \"./packages/*\" dev",
    "build": "pnpm --filter \"./packages/*\" build",
    "template:bot-a": "pnpm --filter bot-a template",
    "template:bot-b": "pnpm --filter bot-b template"
  },
  "devDependencies": {
    "@karinjs/template-react": "workspace:*",
    "typescript": "^5.0.0"
  }
}
```

## 共享组件

### 定义共享组件

```tsx title="packages/shared/components/Avatar.tsx" twoslash
import React from 'react'

export interface AvatarProps {
  src: string
  alt: string
  size?: number
}

export const Avatar: React.FC<AvatarProps> = ({ src, alt, size = 48 }) => (
  <img src={src} alt={alt} className="rounded-full object-cover" style={{ width: size, height: size }} />
)
```

```ts title="packages/shared/components/index.ts"
export { Avatar } from './Avatar'
export { Badge } from './Badge'

```

### 在模板中使用

```tsx title="packages/shared/templates/user/profile/index.tsx" twoslash
// @filename: shared-components.d.ts
declare module '@workspace/shared/components' {
  import type { FC, ReactNode } from 'react'
  export const Avatar: FC<{ src: string; alt: string; size?: number }>
  export const Badge: FC<{ children?: ReactNode }>
}
// @filename: index.tsx
// ---cut---
import { defineTemplate, type TemplateProps } from '@karinjs/template-react'
import { Avatar, Badge } from '@workspace/shared/components'

interface ProfileData {
  name: string
  avatar: string
  level: number
}

const Profile = ({ data }: TemplateProps<ProfileData>) => (
  <div className="flex items-center gap-4 p-6 bg-surface rounded-xl">
    <Avatar src={data.avatar} alt={data.name} size={64} />
    <div>
      <h2 className="text-xl font-bold">{data.name}</h2>
      <Badge>Lv.{data.level}</Badge>
    </div>
  </div>
)

export default defineTemplate({
  name: '用户资料',
  component: Profile
})
```

## 多包构建策略

### 并行构建

```json title="package.json"
{
  "scripts": {
    "build": "pnpm --parallel --filter \"./packages/*\" build"
  }
}
```

### 依赖顺序构建

```json title="package.json"
{
  "scripts": {
    "build": "pnpm --filter \"@workspace/types\" build && pnpm --filter \"@workspace/components\" build && pnpm --filter \"./packages/bot-*\" build"
  }
}
```

### Turborepo 集成

```json title="turbo.json"
{
  "$schema": "https://turbo.build/schema.json",
  "pipeline": {
    "build": {
      "dependsOn": ["^build"],
      "outputs": ["dist/**", "lib/**"]
    },
    "dev": {
      "cache": false,
      "persistent": true
    },
    "template": {
      "cache": false
    }
  }
}
```

```json title="package.json"
{
  "scripts": {
    "build": "turbo run build",
    "dev": "turbo run dev --parallel",
    "template": "turbo run template --filter bot-a"
  }
}
```

### 独立构建配置

每个包使用独立的构建插件：

```ts title="packages/bot-a/vite.config.ts" twoslash
import { defineConfig } from 'vite'
import { ktrBuildPlugin } from '@karinjs/template-react/plugin'

export default defineConfig({
  plugins: [ktrBuildPlugin()],
  build: {
    outDir: 'lib',
    lib: {
      entry: 'src/index.ts',
      formats: ['es']
    }
  }
})
```

```ts title="packages/bot-genshin/karin.template.ts" twoslash
import { defineConfig } from '@karinjs/template-react'
// ---cut---
export default defineConfig({
  dir: {
    template: 'ktr/template' // 仅扫描本地模板
  },
  dev: { port: 5181 },
  extraStylePaths: [
    '../../base/templates/style.css' // 引入基础样式
  ]
})
```

在渲染器中同时加载两个注册表：

```ts title="packages/bot-genshin/src/utils/render.ts" twoslash
import { createRenderer, loadTemplateRegistry } from '@karinjs/template-react'

async function createCombinedRenderer() {
  // 加载基础模板（返回路由映射本身）
  const baseRegistry = await loadTemplateRegistry({ root: '../../base' })

  // 加载本地模板
  const localRegistry = await loadTemplateRegistry({ root: '.' })

  // 合并注册表
  const templates = { ...baseRegistry, ...localRegistry }

  return createRenderer(templates, {
    cssPath: 'lib/style.css', // 打包产物里的 CSS（产物目录随各自打包配置）
    outputDir: 'temp/html'
  })
}

export const render = await createCombinedRenderer()
```

## 最佳实践

### 1. 目录组织

```
推荐：按功能分包
✅ packages/ui, packages/themes, packages/templates

避免：过度拆分
❌ packages/avatar, packages/badge, packages/card
```

### 2. 依赖管理

使用 workspace 协议避免版本冲突：

```json
{
  "dependencies": {
    "@workspace/shared": "workspace:*" // 始终使用最新
  }
}
```

### 3. 构建缓存

利用 Turborepo 或 Nx 缓存构建结果：

```json title="turbo.json"
{
  "pipeline": {
    "build": {
      "outputs": ["dist/**", "lib/**"],
      "dependsOn": ["^build"]
    }
  }
}
```

### 4. 类型检查

根目录统一类型检查：

```json title="tsconfig.json"
{
  "references": [{ "path": "./packages/shared" }, { "path": "./packages/bot-a" }, { "path": "./packages/bot-b" }]
}
```

```bash
pnpm tsc --build
```
